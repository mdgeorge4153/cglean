\begin{definition}
  \label{def:orientationSystem}
  \lean{CGLean.OrientationSystem}
  \leanok
  An orientation system on a set $P$ is a function
  $\orient : P^3 → \{0, 1, -1\}$ such that, for all $p, q, r, s, t ∈ P$:
  \begin{itemize}
    \item $\orient(q, r, p) = \orient(p, q, r)$
    \item $\orient(p, r, q) = -\orient(p, q, r)$
    \item if $\orient(t, q, r)$, $\orient(p, t, r)$ and $\orient(p, q, t)$ are
      all $\geq 0$ (resp.\ all $= 1$), then so is $\orient(p, q, r)$
    \item if $\orient(t, s, p) \geq 0$, $\orient(t, s, q) = 1$,
      $\orient(t, s, r) \geq 0$, $\orient(t, p, q) \geq 0$ and
      $\orient(t, q, r) \geq 0$, then $\orient(t, p, r) \geq 0$; and likewise
      with every hypothesis and the conclusion $= 1$
  \end{itemize}
